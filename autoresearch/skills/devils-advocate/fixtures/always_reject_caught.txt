## Scenario: Gate with TRIP + NON-TRIP tests (always-reject mutation should be caught)

### Gate Code (spread_gate.rs)

```rust
pub struct SpreadGateInput {
    pub spread_bps: f64,
    pub max_spread_bps: f64,
    pub trading_mode: TradingMode,
}

pub fn evaluate_spread_gate(input: &SpreadGateInput) -> GateResult {
    if input.trading_mode == TradingMode::Kill {
        return GateResult::Reject(RejectReason::KillMode);
    }
    if input.spread_bps > input.max_spread_bps {
        return GateResult::Reject(RejectReason::SpreadTooWide);
    }
    GateResult::Pass
}
```

### Test Suite (test_spread_gate.rs)

```rust
#[test]
fn test_spread_gate_trip_wide_spread_rejects() {
    // TRIP: spread exceeds max → must reject
    let input = SpreadGateInput {
        spread_bps: 50.0,
        max_spread_bps: 30.0,
        trading_mode: TradingMode::Active,
    };
    let result = evaluate_spread_gate(&input);
    assert_eq!(result, GateResult::Reject(RejectReason::SpreadTooWide));
}

#[test]
fn test_spread_gate_trip_kill_mode_rejects() {
    // TRIP: Kill mode → must reject regardless of spread
    let input = SpreadGateInput {
        spread_bps: 5.0,
        max_spread_bps: 30.0,
        trading_mode: TradingMode::Kill,
    };
    let result = evaluate_spread_gate(&input);
    assert_eq!(result, GateResult::Reject(RejectReason::KillMode));
}

#[test]
fn test_spread_gate_non_trip_healthy_spread_passes() {
    // NON-TRIP: spread within limit + Active mode → must pass
    let input = SpreadGateInput {
        spread_bps: 10.0,
        max_spread_bps: 30.0,
        trading_mode: TradingMode::Active,
    };
    let result = evaluate_spread_gate(&input);
    assert_eq!(result, GateResult::Pass);
}

#[test]
fn test_spread_gate_non_trip_reduce_only_narrow_spread_passes() {
    // NON-TRIP: ReduceOnly + narrow spread → must pass (spread gate doesn't block ReduceOnly)
    let input = SpreadGateInput {
        spread_bps: 5.0,
        max_spread_bps: 30.0,
        trading_mode: TradingMode::ReduceOnly,
    };
    let result = evaluate_spread_gate(&input);
    assert_eq!(result, GateResult::Pass);
}
```
