### Gate Code (risk_state_gate.rs)

```rust
pub struct RiskStateInput {
    pub risk_state: RiskState,
    pub intent_class: IntentClass,
}

pub fn evaluate_risk_state_gate(input: &RiskStateInput) -> GateResult {
    match input.risk_state {
        RiskState::Healthy => GateResult::Pass,
        RiskState::Degraded => {
            if input.intent_class == IntentClass::Close {
                GateResult::Pass  // Allow risk-reducing trades in Degraded
            } else {
                GateResult::Reject(RejectReason::DegradedRiskState)
            }
        }
        RiskState::Maintenance | RiskState::Kill => {
            GateResult::Reject(RejectReason::RiskKill)
        }
    }
}
```
