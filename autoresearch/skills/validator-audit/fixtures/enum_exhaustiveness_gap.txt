## Validator: risk_level_rules.py

```python
from enum import Enum
from typing import Optional

class RiskLevel(Enum):
    LOW = "LOW"
    MED = "MED"
    HIGH = "HIGH"
    CRITICAL = "CRITICAL"

class ValidationResult:
    def __init__(self):
        self.findings: list[str] = []
        self.passed: bool = True

def validate_risk_assessment(report: dict) -> ValidationResult:
    """Validate that a risk assessment report has proper controls for its risk level."""
    result = ValidationResult()
    level = RiskLevel(report.get("risk_level", "LOW"))

    if level == RiskLevel.LOW:
        # LOW risk: basic checks only
        if "reviewer" not in report:
            result.findings.append("LOW risk requires at least one reviewer")
            result.passed = False

    elif level == RiskLevel.MED:
        # MED risk: reviewer + test evidence
        if "reviewer" not in report:
            result.findings.append("MED risk requires reviewer")
            result.passed = False
        if "test_evidence" not in report:
            result.findings.append("MED risk requires test evidence")
            result.passed = False

    elif level == RiskLevel.HIGH:
        # HIGH risk: reviewer + test evidence + sign-off
        if "reviewer" not in report:
            result.findings.append("HIGH risk requires reviewer")
            result.passed = False
        if "test_evidence" not in report:
            result.findings.append("HIGH risk requires test evidence")
            result.passed = False
        if "sign_off" not in report:
            result.findings.append("HIGH risk requires sign-off from lead")
            result.passed = False

    # NOTE: No branch for RiskLevel.CRITICAL
    # A CRITICAL risk report passes validation with zero checks

    return result
```
