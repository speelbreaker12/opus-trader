from enum import Enum
from typing import Optional
from dataclasses import dataclass

class OrderSide(Enum):
    BUY = "BUY"
    SELL = "SELL"

class OrderType(Enum):
    LIMIT = "LIMIT"
    MARKET = "MARKET"
    STOP_LIMIT = "STOP_LIMIT"

class TimeInForce(Enum):
    GTC = "GTC"     # Good til cancelled
    IOC = "IOC"     # Immediate or cancel
    FOK = "FOK"     # Fill or kill

@dataclass
class OrderRequest:
    instrument_id: str
    side: OrderSide
    order_type: OrderType
    quantity: float
    price: Optional[float]
    stop_price: Optional[float]
    time_in_force: TimeInForce
    client_order_id: str

def validate_order_request(req: OrderRequest) -> dict:
    """Comprehensive order request validator with full enum and field coverage."""
    findings = []

    # Rule 1: instrument_id must be non-empty
    if not req.instrument_id or not req.instrument_id.strip():
        findings.append({"rule": "R1", "field": "instrument_id", "msg": "must be non-empty"})

    # Rule 2: side must be valid enum (enforced by type, but check runtime)
    if not isinstance(req.side, OrderSide):
        findings.append({"rule": "R2", "field": "side", "msg": "invalid side"})

    # Rule 3: order_type — exhaustive handling
    if req.order_type == OrderType.LIMIT:
        if req.price is None or req.price <= 0:
            findings.append({"rule": "R3a", "field": "price", "msg": "LIMIT order requires positive price"})
    elif req.order_type == OrderType.MARKET:
        if req.price is not None:
            findings.append({"rule": "R3b", "field": "price", "msg": "MARKET order must not set price"})
    elif req.order_type == OrderType.STOP_LIMIT:
        if req.price is None or req.price <= 0:
            findings.append({"rule": "R3c", "field": "price", "msg": "STOP_LIMIT requires positive price"})
        if req.stop_price is None or req.stop_price <= 0:
            findings.append({"rule": "R3d", "field": "stop_price", "msg": "STOP_LIMIT requires positive stop_price"})
    else:
        # Fail-closed: unknown order type is rejected
        findings.append({"rule": "R3e", "field": "order_type", "msg": f"unknown order type: {req.order_type}"})

    # Rule 4: quantity must be positive
    if req.quantity <= 0:
        findings.append({"rule": "R4", "field": "quantity", "msg": "quantity must be positive"})

    # Rule 5: time_in_force — exhaustive handling
    if req.time_in_force == TimeInForce.GTC:
        pass  # No additional constraints
    elif req.time_in_force == TimeInForce.IOC:
        pass  # No additional constraints
    elif req.time_in_force == TimeInForce.FOK:
        pass  # No additional constraints
    else:
        # Fail-closed: unknown TIF rejected
        findings.append({"rule": "R5", "field": "time_in_force", "msg": f"unknown time_in_force: {req.time_in_force}"})

    # Rule 6: client_order_id must be non-empty and alphanumeric
    if not req.client_order_id or not req.client_order_id.replace("-", "").replace("_", "").isalnum():
        findings.append({"rule": "R6", "field": "client_order_id", "msg": "must be non-empty alphanumeric (hyphens/underscores allowed)"})

    # Rule 7: Cross-field — stop_price only valid for STOP_LIMIT
    if req.order_type != OrderType.STOP_LIMIT and req.stop_price is not None:
        findings.append({"rule": "R7", "field": "stop_price", "msg": "stop_price only valid for STOP_LIMIT orders"})

    passed = len(findings) == 0
    return {
        "passed": passed,
        "findings": findings,
        "fields_checked": [
            "instrument_id", "side", "order_type", "quantity",
            "price", "stop_price", "time_in_force", "client_order_id"
        ]
    }
