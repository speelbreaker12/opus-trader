## Validator: audit_report_validator.py

```python
from dataclasses import dataclass
from typing import Optional

@dataclass
class AuditReport:
    story_id: str
    verdict: str            # "PROVEN" | "DISPROVEN" | "INCONCLUSIVE"
    pass_result: bool       # True = story passes, False = story fails
    evidence_path: str
    reviewer: str
    timestamp: str

def validate_audit_report(report: AuditReport) -> dict:
    """Validate an audit report before it is recorded as official."""
    findings = []

    # Rule 1: story_id must be non-empty
    if not report.story_id:
        findings.append("story_id is empty")

    # Rule 2: verdict must be one of allowed values
    if report.verdict not in ("PROVEN", "DISPROVEN", "INCONCLUSIVE"):
        findings.append(f"Invalid verdict: {report.verdict}")

    # Rule 3: pass_result must be present (it's a bool, always present)
    # This checks field PRESENCE but not field VALUE consistency
    if report.pass_result is None:
        findings.append("pass_result is missing")

    # Rule 4: evidence_path must be non-empty
    if not report.evidence_path:
        findings.append("evidence_path is empty")

    # Rule 5: reviewer must be non-empty
    if not report.reviewer:
        findings.append("reviewer is empty")

    # Rule 6: timestamp must be non-empty
    if not report.timestamp:
        findings.append("timestamp is empty")

    # BUG: No cross-field invariant check!
    # The following adversarial input passes validation:
    #   verdict="DISPROVEN", pass_result=True
    # A disproven story should NEVER have pass_result=True.
    #
    # Similarly:
    #   verdict="PROVEN", pass_result=False
    # is suspicious but allowed.
    #
    # The validator checks field PRESENCE (paper compliance)
    # but not field VALUE CONSISTENCY (semantic compliance).

    passed = len(findings) == 0
    return {"passed": passed, "findings": findings}
```
