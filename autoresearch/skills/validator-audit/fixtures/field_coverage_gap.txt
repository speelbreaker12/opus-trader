## Validator: deployment_manifest_validator.py

```python
from typing import Any

# Schema defines 10 fields for a deployment manifest
MANIFEST_SCHEMA = {
    "service_name":      {"type": "string",  "required": True},
    "version":           {"type": "string",  "required": True},
    "environment":       {"type": "string",  "required": True},
    "replicas":          {"type": "int",     "required": True},
    "health_check_url":  {"type": "string",  "required": True},
    "rollback_version":  {"type": "string",  "required": True},
    "resource_limits":   {"type": "object",  "required": True},
    "dependencies":      {"type": "list",    "required": True},
    "feature_flags":     {"type": "object",  "required": False},
    "canary_percentage": {"type": "float",   "required": False},
}

def validate_manifest(manifest: dict[str, Any]) -> dict:
    """Validate a deployment manifest against the schema."""
    findings = []

    # Rule 1: service_name must be non-empty string
    if not manifest.get("service_name"):
        findings.append("service_name is missing or empty")

    # Rule 2: version must match semver
    version = manifest.get("version", "")
    if not _is_semver(version):
        findings.append(f"version '{version}' is not valid semver")

    # Rule 3: environment must be one of allowed values
    env = manifest.get("environment", "")
    if env not in ("dev", "staging", "production"):
        findings.append(f"environment '{env}' is not valid")

    # Rule 4: replicas must be positive integer
    replicas = manifest.get("replicas", 0)
    if not isinstance(replicas, int) or replicas < 1:
        findings.append(f"replicas must be positive integer, got {replicas}")

    # Rule 5: health_check_url must start with http
    hc = manifest.get("health_check_url", "")
    if not hc.startswith("http"):
        findings.append("health_check_url must start with http")

    # Rule 6: rollback_version must differ from version
    rb = manifest.get("rollback_version", "")
    if rb == version:
        findings.append("rollback_version must differ from version")

    # MISSING: No rule for 'resource_limits' (required field, never validated)
    # MISSING: No rule for 'dependencies' (required field, never validated)
    # MISSING: No rule for 'feature_flags' (optional but never validated)
    # MISSING: No rule for 'canary_percentage' (optional but never validated)

    passed = len(findings) == 0
    return {"passed": passed, "findings": findings}


def _is_semver(v: str) -> bool:
    import re
    return bool(re.match(r"^\d+\.\d+\.\d+$", v))
```
